import yaml
import re
from PIL import Image
import imagehash

class SemioticVelocityPipeline:
    def __init__(self, config_path):
        """Initializes the v5.3 pipeline with clean configuration variables."""
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
        target = self.config['target_objects'][0]
        self.thresholds = target['thresholds']
        self.anchors = target['linguistic_anchors']

    def process_visual_stream(self, image_path, original_canvas_path):
        """Module 1: Processes images using difference hashing thresholds."""
        try:
            img1 = Image.open(image_path)
            img2 = Image.open(original_canvas_path)
            hash1 = imagehash.dhash(img1)
            hash2 = imagehash.dhash(img2)
            hamming_dist = hash1 - hash2
            if hamming_dist <= self.thresholds['tier_a_hamming']:
                return "Tier A: Direct Reproduction"
            elif hamming_dist <= self.thresholds['tier_b_hamming']:
                return "Tier B: Close Derivative"
            elif hamming_dist <= self.thresholds['tier_c_hamming']:
                return "Tier C: Loose Derivative"
            else:
                return "Out of core hashing bounds. Routing to advanced Tier D SIFT matching."
        except Exception as e:
            return f"Error processing image stream: {str(e)}"

    def process_linguistic_stream(self, text_content):
        """Module 2: Evaluates text tokens using morphology and contextual anchors."""
        text_lower = text_content.lower()
        has_root = any(root.lower() in text_lower for root in self.anchors['roots'])
        has_frag = any(re.search(fr"\b{frag.lower()}\b", text_lower) for frag in self.anchors['fragments'])
        if not (has_root or has_frag):
            return "No motif detected."
        has_primary = any(p.lower() in text_lower for p in self.anchors['primary_anchors'])
        has_secondary = any(s.lower() in text_lower for s in self.anchors['secondary_anchors'])
        is_false_positive = any(ex.lower() in text_lower for ex in self.anchors['exclusion_terms'])
        if (has_primary or has_secondary) and not is_false_positive:
            return "Tier E: Verified Nominal Deployment"
        return "Linguistic match found, but failed contextual anchor safety check."
